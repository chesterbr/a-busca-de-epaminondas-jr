DIM A(M)
IF N = 77 AND A(77) = -999 THEN PRINT "A"
IF N = 77 AND X = -999 THEN PRINT "A"
IF N = 77 AND A$ = "x" THEN PRINT "A"
IF A(77) = -999 AND N = 77 THEN PRINT "A"
IF N = 77 AND A(N) <> X THEN PRINT "A"
IF N = 77 THEN IF A(77) = -999 THEN PRINT "A"
IF N = 77 AND A(77) = -999 THEN PRINT "A" : PRINT "B"
IF N = 77 AND NOT A(77) THEN PRINT "A"
IF N <> 77 OR A(77) <> -999 THEN PRINT "A"
IF N = 77 AND SALA = 4 THEN PRINT "A" : PRINT "B" : RETURN
